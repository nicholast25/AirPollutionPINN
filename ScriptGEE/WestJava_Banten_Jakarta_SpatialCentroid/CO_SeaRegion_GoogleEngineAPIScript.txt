
var provinces = ee.FeatureCollection("FAO/GAUL/2015/level1")
  .filter(ee.Filter.inList('ADM1_NAME', ['Banten', 'Dki Jakarta', 'Jawa Barat']));

var land = provinces.union().geometry();
var seaBuffer = land.buffer(22224); // 22.224 km buffer from land
var surroundingSea = seaBuffer.difference(land, 1); // only sea

Map.centerObject(surroundingSea, 9);
Map.addLayer(surroundingSea, {color: 'blue'}, 'Sea Area');

// Grid size in degrees (~7 km × 5.5 km)  --> Make it 2 x 2 Grid
var dx = 0.1258;  // longitude ~14 km
var dy = 0.0988;  // latitude ~11 km

var bounds = surroundingSea.bounds();
var lonLatRect = bounds.coordinates().get(0);
lonLatRect = ee.List(lonLatRect); // convert to usable EE list

var lons = lonLatRect.map(function(coord) {
  return ee.Number(ee.List(coord).get(0));
});
var lats = lonLatRect.map(function(coord) {
  return ee.Number(ee.List(coord).get(1));
});

var lonMin = ee.Number(lons.reduce(ee.Reducer.min()));
var lonMax = ee.Number(lons.reduce(ee.Reducer.max()));
var latMin = ee.Number(lats.reduce(ee.Reducer.min()));
var latMax = ee.Number(lats.reduce(ee.Reducer.max()));


// Generate lon/lat sequences for grid creation
var lonList = ee.List.sequence(lonMin, lonMax, dx);
var latList = ee.List.sequence(latMin, latMax, dy);

var nestedGrid = lonList.map(function(lon) {
  return latList.map(function(lat) {
    var rect = ee.Geometry.Rectangle([
      [lon, lat],
      [ee.Number(lon).add(dx), ee.Number(lat).add(dy)]
    ]);
    return ee.Feature(rect);
  });
});

var flatGrid = ee.FeatureCollection(ee.List(nestedGrid).flatten());

var seaGrid = flatGrid.filterBounds(surroundingSea).map(function(f) {
  return f.intersection(surroundingSea, 1);
});
Map.addLayer(seaGrid, {color: 'red'}, 'Sea Grid');

var startDate = ee.Date('2023-05-01');
var endDate = ee.Date('2025-04-30');  
var nDays = endDate.difference(startDate, 'day');
var dates = ee.List.sequence(0, nDays.subtract(1)).map(function(d) {
  return startDate.advance(d, 'day');
});

var dailyCollections = dates.map(function(date) {
  date = ee.Date(date);
  var image = ee.ImageCollection('COPERNICUS/S5P/OFFL/L3_CO')
    .select('CO_column_number_density')
    .filterDate(date, date.advance(1, 'day'))
    .mean();

  var zonal = image.reduceRegions({
    collection: seaGrid,
    reducer: ee.Reducer.mean(),
    scale: 10000
  });

  return zonal.map(function(f) {
    var center = f.geometry().centroid().coordinates();
    return f.set({
      'date': date.format('YYYY-MM-dd'),
      'longitude': center.get(0),
      'latitude': center.get(1)
    });
  });
});

var seaCO = ee.FeatureCollection(dailyCollections).flatten();

// 8. Export to Google Drive
Export.table.toDrive({
  collection: seaCO,
  description: 'CO_Sea_4xGrid',
  fileFormat: 'CSV',
  selectors: ['date', 'mean', 'longitude', 'latitude']
});
