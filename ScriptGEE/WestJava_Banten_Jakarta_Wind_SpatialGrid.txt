// ======================================================
// 1. Load level-2 administrative boundaries
// ======================================================

var level2 = ee.FeatureCollection("FAO/GAUL/2015/level2");


// ======================================================
// 2. Filter all cities/regencies inside
//    DKI Jakarta, Banten, and Jawa Barat
// ======================================================

var indonesia = level2.filter(
  ee.Filter.eq('ADM0_NAME', 'Indonesia')
);

var targetProvinces = [Banten', 'Dki Jakarta', 'Jawa Barat'];

var studyArea = indonesia.filter(
  ee.Filter.inList('ADM1_NAME', targetProvinces)
);


// ======================================================
// 3. Merge geometry of study area
// ======================================================

var land = studyArea.union().geometry();

Map.centerObject(land, 8);

Map.addLayer(
  land,
  {color: 'green'},
  'Study Area'
);


// ======================================================
// 4. Grid size (~14 km × ~11 km)
// ======================================================

var dx = 0.1258;   // longitude step
var dy = 0.0988;   // latitude step


// ======================================================
// 5. Create bounding box and generate grid
// ======================================================

var bounds = land.bounds();

var coords = ee.List(bounds.coordinates().get(0));

var lons = coords.map(function(coord) {
  return ee.Number(ee.List(coord).get(0));
});

var lats = coords.map(function(coord) {
  return ee.Number(ee.List(coord).get(1));
});

var lonMin = ee.Number(lons.reduce(ee.Reducer.min()));
var lonMax = ee.Number(lons.reduce(ee.Reducer.max()));

var latMin = ee.Number(lats.reduce(ee.Reducer.min()));
var latMax = ee.Number(lats.reduce(ee.Reducer.max()));

var lonList = ee.List.sequence(lonMin, lonMax, dx);
var latList = ee.List.sequence(latMin, latMax, dy);


// ======================================================
// 6. Create rectangular grid tiles
// ======================================================

var nestedGrid = lonList.map(function(lon) {

  return latList.map(function(lat) {

    var rect = ee.Geometry.Rectangle([
      [lon, lat],
      [ee.Number(lon).add(dx), ee.Number(lat).add(dy)]
    ]);

    return ee.Feature(rect);

  });

});

var flatGrid = ee.FeatureCollection(
  ee.List(nestedGrid).flatten()
);


// ======================================================
// 7. Filter grid to land area and clip
// ======================================================

var landGrid = flatGrid
  .filterBounds(land)
  .map(function(f) {
    return f.intersection(land, 1);
  });


// ======================================================
// 8. Visualization
// ======================================================

Map.addLayer(
  studyArea.style({
    color: 'blue',
    fillColor: '00000000',
    width: 1
  }),
  {},
  'City Boundaries'
);

Map.addLayer(
  landGrid.style({
    color: 'orange',
    fillColor: '00000000',
    width: 1
  }),
  {},
  'Grid Cells'
);


// ======================================================
// 9. Define time range
// ======================================================

var startDate = ee.Date('2023-05-01');
var endDate   = ee.Date('2025-04-30');

var nDays = endDate.difference(startDate, 'day');

var dates = ee.List.sequence(0, nDays.subtract(1)).map(function(d) {
  return startDate.advance(d, 'day');
});


// ======================================================
// 10. Extract mean daily wind speed and direction
// ======================================================

var windCollections = dates.map(function(date) {

  date = ee.Date(date);

  // Daily mean from hourly ERA5-Land data
  var wind = ee.ImageCollection('ECMWF/ERA5_LAND/HOURLY')
    .filterDate(date, date.advance(1, 'day'))
    .select([
      'u_component_of_wind_10m',
      'v_component_of_wind_10m'
    ])
    .mean();

  // Wind components
  var u = wind.select('u_component_of_wind_10m');
  var v = wind.select('v_component_of_wind_10m');

  // Wind speed
  var speed = u.pow(2)
    .add(v.pow(2))
    .sqrt()
    .rename('wind_speed_10m');

  // Wind direction (azimuth)
  var direction = u.atan2(v)
    .multiply(180)
    .divide(Math.PI)
    .add(180)
    .rename('wind_direction_10m');

  // Combine bands
  var combined = wind
    .addBands(speed)
    .addBands(direction);

  // Zonal statistics
  var zonal = combined.reduceRegions({
    collection: landGrid,
    reducer: ee.Reducer.mean(),
    scale: 10000
  });

  // Add metadata
  return zonal.map(function(f) {

    var center = f.geometry()
      .centroid()
      .coordinates();

    return f.set({
      'date': date.format('YYYY-MM-dd'),
      'longitude': center.get(0),
      'latitude': center.get(1)
    });

  });

});


// ======================================================
// 11. Flatten collection
// ======================================================

var landWind = ee.FeatureCollection(
  windCollections
).flatten();


// ======================================================
// 12. Export CSV
// ======================================================

Export.table.toDrive({
  collection: landWind,
  description: 'Wind_MeanDaily_Jakarta_Banten_JawaBarat',
  fileFormat: 'CSV',
  selectors: [
    'date',
    'longitude',
    'latitude',
    'u_component_of_wind_10m',
    'v_component_of_wind_10m',
    'wind_speed_10m',
    'wind_direction_10m'
  ]
});
